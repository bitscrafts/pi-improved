# The gate's ONE step list (spec 146). quality-gate.sh reads it for the local run and for the
# `cargo xtask fresh-gate` export alike; add, remove or edit a step HERE and nowhere else.
# Format: <label>|<shell command>   (one line per step; `#` lines and blank lines are ignored).
# A step passes only if its command exits 0; every step runs even after an earlier failure.

fmt|cargo fmt --all --check

# F-01 fix: clippy's --all-targets build writes libschemars/libserde_json (and other
# proc-macro-generated) rlibs into target/debug/deps with a feature-unification hash that differs from
# the plain `cargo test` build; rustdoc's doctest linker then fails with E0463. Own target dir.
clippy|CARGO_TARGET_DIR="${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}"/clippy-gate cargo clippy --workspace --all-targets -- -D warnings

# No `llamacpp` clippy step (spec 170): it cost ~720s of C++ per run for a binding pii never uses (models are reached over HTTP). Manual check: see _docs/05-operations/deploy-runbook.md.

# Spec 121 (R6): the real GPIO backend (`hal-gpiod`) is compiled and linted so it cannot rot; NEVER run
# (no /dev/gpiochip* here). Own target dir.
clippy (hal-gpiod)|CARGO_TARGET_DIR="${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}"/hal-gate cargo clippy -p pii-hal --features hal-gpiod --all-targets -- -D warnings

# pii-worker's supervisor integration tests spawn the real `pii` binary at target/debug/pii (a fixed
# path), so it must exist before `cargo test` runs.
# The warm cache (FRESH_GATE_TARGET) builds the workspace OUTSIDE the export, so the fixed path the pii-worker
# supervisor tests spawn — target/debug/pii, inside the export — never appears. Publish the binary where they
# look while keeping the warm cache: a 48-minute cold build per gate run is not survivable, and the tests are
# right to spawn the real binary.
build|CARGO_TARGET_DIR="${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}/build-gate" cargo build --workspace && mkdir -p target/debug && cp -f "${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}/build-gate/debug/pii" target/debug/pii

# pii-worker's control tests spawn `pii` from the test executable's own target dir (test-gate/debug/pii).
# The `build` step above writes to build-gate and target/debug, not here, so build pii into test-gate first.
test (lib/bins/tests)|CARGO_TARGET_DIR="${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}"/test-gate cargo build -q -p pii-cli --bin pii && CARGO_TARGET_DIR="${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}"/test-gate cargo test -j 2 --workspace --lib --bins --tests

# F-01: doctests build each lib WITHOUT cfg(test); sharing target/debug/deps with the step above can
# pick a stale rlib hash (E0463). Own CARGO_TARGET_DIR so it always builds its own libs.
test (doc)|CARGO_TARGET_DIR="${FRESH_GATE_TARGET:-/projects/pi-improved/_tmp/gate-target}"/doctest-gate cargo test --workspace --doc

# Spec 108 R2: line coverage of lib/bins/tests (no doctests) must stay at or above coverage-floor.txt.
# Needs cargo-llvm-cov + the llvm-tools component; their absence is a FAIL, not a skip. Own target dir.
# QG_SKIP_COVERAGE=1 skips ONLY this step and says so LOUDLY (a fresh-gate export cannot fit the ~18G
# instrumented build on a small volume). It is NEVER implied: unset means the floor is measured.
coverage-floor|if [ "${QG_SKIP_COVERAGE:-0}" = "1" ]; then echo "SKIPPED: coverage-floor (QG_SKIP_COVERAGE=1 — the floor was NOT measured in this run,"; echo "         the repo gate enforces it by default and this line is the record that it was skipped)"; else cargo llvm-cov --workspace --lib --bins --tests --json --summary-only --output-path target/coverage-summary.json >/dev/null && CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- coverage-floor target/coverage-summary.json; fi

# 450-effective-line file gate (spec 67, Fix 1): pure text scan; also enforced here for pidag runs.
lint-files|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- lint-files

# Spec 96: tempdir_in on a crate-local `_tmp` must be preceded by create_dir_all / tmp_scratch (N1 class).
lint-scratch|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- lint-scratch

# Spec 98: every crate uses the pii-observe seam (or a named exemption), no new raw prints, registry in ALL.
lint-observability|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- lint-observability

# Spec 72 (D3 fix): every declared-capability seed needle has a REAL external caller.
reachability-check|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- reachability-check

# Spec 118: the repo's five dependency invariants. A failure to run is a FAIL, never a skip.
deps-check|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- deps-check

# Spec 118: sockets/name resolution only in the allowlisted files.
net-check|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- net-check

# Spec 118: backticked repo paths in HANDOFF.md must exist. Offline and deterministic.
doc-drift-check|CARGO_TARGET_DIR="_tmp/tools-gate" cargo run -q -p xtask -- doc-drift-check
